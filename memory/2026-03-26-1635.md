```bash
#!/usr/bin/env bash
BASE="/Users/aaron/Documents/Obsidian Vault"

# 作者 ID（上一步创建的 Person 实体的 ID）
AUTHOR_ID="person_001"

# 项目 ID 映射（请根据上一步创建的实际 ID 替换）
PROJ_COMPANY="proj_company"
PROJ_WAIMAI="proj_waimai"
PROJ_ZHIHUIKU="proj_zhihui"
PROJ_ROOT="proj_root"

# 创建文档函数
create_doc() {
  local file_path="$1"
  local folder_name="$2"
  local title="$(basename "$file_path" .md)"
  # 创建 Document 实体
  python3 scripts/ontology.py create --type Document --props "{
    \"title\":\"$title\",
    \"path\":\"$file_path\",
    \"summary\":\"\"
  }"
  # 取得刚创建的实体 ID（脚本会在 stdout 打印 ID，假设为 $DOC_ID）
  DOC_ID=$(python3 scripts/ontology.py last-id)   # 这里需要你自行实现或手动复制
  # 关联到项目
  case "$folder_name" in
    Company)   REL_PROJ=$PROJ_COMPANY ;;
    外卖)      REL_PROJ=$PROJ_WAIMAI ;;
    知识库)    REL_PROJ=$PROJ_ZHIHUIKU ;;
    *)         REL_PROJ=$PROJ_ROOT ;;
  esac
  python3 scripts/ontology.py relate --from "$DOC_ID" --rel belongs_to --to "$REL_PROJ"
  # 关联作者
  python3 scripts/ontology.py relate --from "$DOC_ID" --rel authored_by --to "$AUTHOR_ID"
}

# 遍历 markdown 文件
while IFS= read -r -d '' file; do
  # 取父目录的顶层文件夹名
  top_folder=$(echo "$file" | awk -F'/' '{print $(NF-1)}')
  create_doc "$file" "$top_folder"
done < <(find "$BASE" -type f -name "*.md" -print0)
```

> 说明  
> * `python3 scripts/ontology.py last-id` 需要你在 `ontology.py` 中实现一个返回最近一次 `create` 返回的实体 ID 的小函数（如果没有现成的，可手动在创建后记录）。若你不想自行改脚本，也可以把每条 `create`‑`relate` 命令单独运行，我可以帮你逐条执行。
